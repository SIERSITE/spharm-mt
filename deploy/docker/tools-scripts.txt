# deploy/docker/tools-scripts.txt
#
# Comandos do package.json que TÊM de funcionar dentro da imagem
# `migrator` — a que corre no perfil `tools` do compose:
#
#   docker compose --profile tools run --rm migrate npm run <comando>
#
# Esta lista é a fonte de verdade da auditoria. O build da imagem resolve
# cada nome no package.json, extrai o ficheiro de entrada e FALHA se
# algum não existir na imagem (ver audit-tools-entrypoints.mjs).
#
# Porquê uma lista explícita e não «tudo o que está no package.json»:
# o package.json tem ~60 comandos e a maioria são de desenvolvimento ou
# ad-hoc de investigação. Copiar tudo para a imagem operacional seria
# arrastar código que ninguém corre em produção. A lista diz, de forma
# auditável, o que é operacional — e o build obriga a mantê-la honesta.
#
# A falha que motivou este ficheiro:
#
#   docker compose run --rm migrate npm run tenant:create
#   ERR_MODULE_NOT_FOUND: /app/scripts/admin/create-client.ts
#
# O comando oficial de criação de tenants apontava para `scripts/admin`,
# que o Dockerfile não copiava. Nada no build acusava isso: descobria-se
# ao correr o comando, na VPS, no momento de criar o primeiro cliente.
#
# Ao acrescentar um comando aqui, garantir que o Dockerfile copia o
# directório correspondente para o estágio `migrator`. O build diz
# exactamente qual falta.

# ── Migrations ───────────────────────────────────────────────────────
# Sem ficheiro próprio (é o CLI do Prisma) — auditado como presente.
db:migrate:deploy

# ── Control plane ────────────────────────────────────────────────────
control:generate
control:migrate:deploy
control:migrate:status
control:create-global-admin

# ── Tenants ──────────────────────────────────────────────────────────
tenancy:provision
tenancy:list
tenancy:migrate-all
tenancy:health
tenancy:deactivate
tenancy:reactivate
tenancy:issue-ingest-key
tenancy:cleanup-failed
tenancy:smoke-resolver
tenancy:debug-ingest-auth
tenancy:status
tenancy:add-farmacia
tenancy:add-user
tenant:onboard

# ── Criação de cliente ───────────────────────────────────────────────
# `tenant:create` e `tenancy:create` são o MESMO ficheiro. Os dois nomes
# circulam na documentação de onboarding, portanto os dois são auditados.
tenant:create
tenancy:create

# ── Catálogo ─────────────────────────────────────────────────────────
# O bootstrap do catálogo de um tenant, por ORDEM DE EXECUÇÃO. Correm no
# servidor, contra o PostgreSQL local, e o destino é sempre explícito:
# `--tenant=<slug>` traz base, host e credencial do control plane. Não há
# base por omissão, e o `--db=<base>` que sobrou é só para
# desenvolvimento — em produção usa o utilizador do DATABASE_URL, que não
# tem CONNECT às bases dos tenants (ver lib/catalog/target-db.ts).
#
# Todos aceitam `--dry-run`, e em dry-run abrem a sessão PostgreSQL em
# read-only.
catalog:seed-utilizacoes
catalog:classify-backfill
catalog:fill-rules
catalog:backfill-utilizacoes
# Knowledge-enrichment: a fase que usa modelo. Sem `--apply` não escreve
# nada, nem cache; com `--apply` custa dinheiro por produto.
catalog:knowledge-enrich

# ── Operação ─────────────────────────────────────────────────────────
admin:reset-user-password
env:doctor

# ── Pacote do agent ──────────────────────────────────────────────────
# ATENÇÃO ao que esta linha garante e ao que não garante. A auditoria
# prova que o CÓDIGO está na imagem — o entrypoint e todos os módulos
# que ele importa. Não prova que corre: o `package-agent` copia de
# `dist-agent/SPharmMT-Agent/`, um directório de BUILD que a imagem não
# tem. Para o usar no servidor é preciso extrair lá o ZIP base:
#
#   unzip /opt/spharmmt/agent-base/spharmmt-agent-base.zip \
#         -d <dir>/dist-agent/SPharmMT-Agent
#
# O caminho normal do Admin Wizard NÃO passa por aqui: o wizard pede o
# `agent-package` à API (que só emite a chave e devolve o config) e monta
# o ZIP na máquina do técnico, a partir do template servido em
# /agent-base/.
admin:package-agent

# ── Fora desta lista, de propósito ───────────────────────────────────
# agent:*         — corre na farmácia, não no servidor
# admin-wizard:*  — PowerShell, corre na máquina do operador
# catalog:simulate, catalog:audit, catalog:export/import — investigação
#                   do catálogo, correm na máquina de quem analisa
# ingest:*        — manutenção pontual de dados, não operação da stack
# test:*, lint, typecheck, dev, build, start — desenvolvimento
# scheduler       — corre no `runner` (serviço worker), não no migrator
